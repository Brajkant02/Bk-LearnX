MySQL/SQL - Part 3

SELECT, WHERE, DISTINCT, ORDER BY, GROUP BY, HAVING, operators और JOINs को deep practical format में समझाया गया है।

SELECT Command

SELECT * FROM student;
SELECT name, marks FROM student;

SELECT required columns और rows retrieve करता है।

WHERE Clause

SELECT * FROM student
WHERE marks > 70;

WHERE condition के आधार पर rows filter करता है।

DISTINCT Clause

SELECT DISTINCT city FROM student;

Duplicate values remove करके unique values show करता है।

ORDER BY

SELECT * FROM student
ORDER BY marks DESC;

ASC ascending और DESC descending sorting के लिए use होते हैं।

GROUP BY and HAVING

SELECT city, COUNT(*) AS total_students
FROM student
GROUP BY city
HAVING COUNT(*) > 2;

GROUP BY groups बनाता है और HAVING grouped result filter करता है।

Logical Operators

SELECT * FROM student
WHERE marks > 70 AND city='Kanpur';

SELECT * FROM student
WHERE city='Kanpur' OR city='Delhi';

SELECT * FROM student
WHERE NOT city='Lucknow';

Introduction to JOIN

JOIN related columns के आधार पर multiple tables का data combine करता है।

Image Placeholder: SQL Join Concept

INNER JOIN

SELECT s.name, d.dept_name
FROM student s
INNER JOIN department d
ON s.dept_id = d.dept_id;

Only matching rows return होती हैं।

LEFT JOIN

SELECT s.name, d.dept_name
FROM student s
LEFT JOIN department d
ON s.dept_id = d.dept_id;

Left table की सभी rows और matching right rows return होती हैं।

RIGHT JOIN

SELECT s.name, d.dept_name
FROM student s
RIGHT JOIN department d
ON s.dept_id = d.dept_id;

CROSS JOIN and SELF JOIN

CROSS JOIN

SELECT * FROM student CROSS JOIN course;

SELF JOIN

SELECT e.name AS employee,
       m.name AS manager
FROM employee e
LEFT JOIN employee m
ON e.manager_id = m.employee_id;
